import { sign, verify } from 'jsonwebtoken';

interface TokenPayload {
  userId: string;
  role: string;
  exp?: number;
}

export class TokenService {
  private secretKey = process.env.JWT_SECRET;

  generateToken(payload: TokenPayload): string {
    return sign(payload, this.secretKey, {
      expiresIn: '1h'
    });
  }
}
